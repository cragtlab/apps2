(async function() {
    const title = "Stumble";
    const container = PluginAPI.add(title);
    const tabId = "tab_" + title.replace(/\s+/g, "_");

    container.id = tabId;

    const style = document.createElement('style');
    style.textContent = `

        #${tabId} .btn-danger { background: #ff4d4d; color: white; }
        #${tabId} .btn-snooze { background: orange; color: white; }
        #${tabId} .btn-add { background: #28a745; color: white; }
        #${tabId} #linkFreqContainer { display: none; gap: 10px; margin-bottom: 10px; }

        }

        }

            display: block;
            overflow-wrap: anywhere;
        }

    document.head.appendChild(style);

    const STORAGE_KEY = 'stumble_links';
    const DATA_FILE_NAME = 'stumbleBak.txt';

    let links = [];
    let currentLink = null;
    let needsFileSave = false;
    let isSavingToFile = false;

    container.innerHTML = `
        <div class="container">
            <b>Stumble Links - (open tab group in mobile)</b>
            <button id="btnLoadFromFile" class="btn-add" style="padding: 4px 8px; font-size: 11px; margin-left: 10px;">Load from file</button>

            <!-- DISPLAY -->
            <div id="linkFreqContainer">
                <input id="frequencyInput" placeholder="e.g. 2, W, M, Y">
                <a id="linkDisplay" target="_blank" style="flex:1;"></a>
            </div>

            <div id="titleEditorContainer" style="display:none;">
                <textarea id="titleTextarea" placeholder="enter remarks"></textarea>
            </div>

            <div id="buttonContainer" style="display:none;">
                <button id="btn-start" class="btn-add" style="display:none;">Start</button>
                <button id="btn-snooze" class="btn-snooze">Snooze</button>
                <button id="btn-delete" class="btn-danger">Delete</button>
                <button id="btn-update-url">Update URL only</button>
            </div>

            <!-- CONTROLS -->
            <hr>
            <div>
                <button id="btnAddFromClipboard" class="btn-add">Add</button>
                <button id="btnOpenReflect">R</button>
                <button id="btnOpen10x">10x</button>
                <input type="search"
                    id="searchInput"
                    placeholder="search..."
                    style="padding:8px;width:140px;"
                >
            </div>

            <!-- TABLE -->
            <table>
                <thead>
                    <tr>
                        <th>Snoozed</th>
                        <th>Freq</th>
                        <th>Link</th>
                    </tr>
                </thead>
                <tbody id="linksTableBody"></tbody>
            </table>
        </div>
    `;

    const btnLoadFromFile = container.querySelector('#btnLoadFromFile');
    const linkFreqContainer = container.querySelector('#linkFreqContainer');
    const frequencyInput = container.querySelector('#frequencyInput');
    const linkDisplay = container.querySelector('#linkDisplay');
    const titleEditorContainer = container.querySelector('#titleEditorContainer');
    const titleTextarea = container.querySelector('#titleTextarea');
    const buttonContainer = container.querySelector('#buttonContainer');
    const btnStart = container.querySelector('#btn-start');
    const btnSnooze = container.querySelector('#btn-snooze');
    const btnDelete = container.querySelector('#btn-delete');
    const btnUpdateUrl = container.querySelector('#btn-update-url');
    const btnAddFromClipboard = container.querySelector('#btnAddFromClipboard');
    const btnOpenReflect = container.querySelector('#btnOpenReflect');
    const btnOpen10x = container.querySelector('#btnOpen10x');
    const searchInput = container.querySelector('#searchInput');
    const linksTableBody = container.querySelector('#linksTableBody');

    function adjustTextareaHeight() {
        if (!titleTextarea) return;
        titleTextarea.style.height = 'auto';
        titleTextarea.style.height = titleTextarea.scrollHeight + 'px';
    }

    async function init() {
        await loadLinks();

        displayNextAvailable();
        updateLinksList();

        if (titleTextarea) {
            titleTextarea.addEventListener('input', adjustTextareaHeight);
        }

        if (getAvailableLinks().length > 0) {
            if (btnStart) btnStart.style.display = 'inline-block';
        }
    }

    async function loadFromFile() {
        try {
            const rootHandle = PluginAPI.getRootHandle();
            if (!rootHandle) {
                alert('Please load folder/plugins first.');
                return;
            }
            const fileHandle = await rootHandle.getFileHandle(DATA_FILE_NAME, { create: false });
            const file = await fileHandle.getFile();
            const text = await file.text();
            if (!text || !text.trim()) {
                alert('File stumbleBak.txt is empty.');
                return;
            }
            const loadedLinks = JSON.parse(text);
            if (!Array.isArray(loadedLinks)) {
                alert('Invalid JSON format in stumbleBak.txt.');
                return;
            }
            links = loadedLinks.map(link => ({
                ...link,
                snoozed_until: normalizeSnoozedUntil(link.snoozed_until)
            }));
            localStorage.setItem(STORAGE_KEY, JSON.stringify(links));
            needsFileSave = false;
            displayNextAvailable();
            updateLinksList();
            alert(`Loaded ${links.length} records from file.`);
        } catch (err) {
            console.error('Error loading from file:', err);
            alert('Failed to load from file: ' + err.message);
        }
    }

    async function loadLinks() {
        let loadedFromLocal = false;
        const localData = localStorage.getItem(STORAGE_KEY);
        if (localData) {
            try {
                links = JSON.parse(localData);
                loadedFromLocal = true;
            } catch (e) {
                console.error('Error parsing localStorage:', e);
            }
        }

        if (!loadedFromLocal) {
            try {
                const rootHandle = PluginAPI.getRootHandle();
                if (rootHandle) {
                    const fileHandle = await rootHandle.getFileHandle(DATA_FILE_NAME, { create: false });
                    const file = await fileHandle.getFile();
                    const text = await file.text();
                    if (text && text.trim()) {
                        links = JSON.parse(text);
                        saveLinks();
                    }
                }
            } catch (e) {
                // File not found or read error
            }
        }

        const normalizedLinks = links.map(link => ({
            ...link,
            snoozed_until: normalizeSnoozedUntil(link.snoozed_until)
        }));

        const changed = normalizedLinks.some(
            (link, index) => link.snoozed_until !== links[index]?.snoozed_until
        );

        links = normalizedLinks;

        if (changed) {
            saveLinks();
        }
    }

    function saveLinks(emptyExpected = false) {
        if (links.length === 0 && !emptyExpected) {
            return;
        }

        localStorage.setItem(
            STORAGE_KEY,
            JSON.stringify(links)
        );
        needsFileSave = true;
    }

    async function saveToFile() {
        if (!needsFileSave || isSavingToFile) return;
        isSavingToFile = true;
        try {
            const rootHandle = PluginAPI.getRootHandle();
            if (rootHandle) {
                const fileHandle = await rootHandle.getFileHandle(DATA_FILE_NAME, { create: true });
                const writable = await fileHandle.createWritable();
                await writable.write(JSON.stringify(links, null, 2));
                await writable.close();
                needsFileSave = false;
            }
        } catch (err) {
            console.error('Failed to save to stumbleBak.txt:', err);
        } finally {
            isSavingToFile = false;
        }
    }

    function getLocalDateKey(date = new Date()) {
        const year = date.getFullYear();
        const month = String(date.getMonth() + 1).padStart(2, '0');
        const day = String(date.getDate()).padStart(2, '0');
        return `${year}-${month}-${day}`;
    }

    function normalizeSnoozedUntil(value) {
        if (value === null || value === undefined || value === '') {
            return null;
        }

        if (typeof value === 'string') {
            const trimmed = value.trim();
            if (!trimmed) {
                return null;
            }
            if (/^\d{4}-\d{2}-\d{2}$/.test(trimmed)) {
                return trimmed;
            }
            if (/^\d+$/.test(trimmed)) {
                value = Number(trimmed);
            } else {
                const parsed = new Date(trimmed);
                return Number.isNaN(parsed.getTime())
                    ? trimmed
                    : getLocalDateKey(parsed);
            }
        }

        if (typeof value === 'number') {
            return Number.isFinite(value)
                ? getLocalDateKey(new Date(value))
                : null;
        }

        if (value instanceof Date) {
            return Number.isNaN(value.getTime())
                ? null
                : getLocalDateKey(value);
        }

        return null;
    }

    function getAvailableLinks() {
        const today = getLocalDateKey();
        return links.filter(
            l => !l.snoozed_until || l.snoozed_until <= today
        );
    }

    function displayNextAvailable() {
        const available = getAvailableLinks();

        if (!available.length) {
            currentLink = null;
            linkFreqContainer.style.display = 'none';
            titleEditorContainer.style.display = 'none';
            buttonContainer.style.display = 'none';
            return;
        }

        displayLink(available[0]);
    }

    function displayLink(link) {
        if (!link) return;

        currentLink = link;

        linkFreqContainer.style.display = 'flex';
        titleEditorContainer.style.display = 'block';
        buttonContainer.style.display = 'block';

        linkDisplay.href = link.link;
        linkDisplay.textContent = link.link;

        titleTextarea.value = link.title || '';
        adjustTextareaHeight();

        frequencyInput.value = link.frequency || '2';
    }

    function saveCurrent() {
        if (!currentLink) return;

        currentLink.title = titleTextarea.value;
        currentLink.frequency = frequencyInput.value || '2';
    }

    async function updateCurrentURL() {
        if (!currentLink) return;

        const draftTitle = titleTextarea.value;
        const draftFreq = frequencyInput.value;

        let clipboardText = '';
        try {
            clipboardText = (await navigator.clipboard.readText()).trim();
        } catch (e) {}

        let newUrl = clipboardText;

        if (
            !newUrl ||
            (
                !newUrl.startsWith('http') &&
                !newUrl.startsWith('file')
            )
        ) {
            newUrl = '';
        }

        const input = prompt(
            'Update URL:',
            newUrl || currentLink.link
        );

        if (input === null) {
            titleTextarea.value = draftTitle;
            frequencyInput.value = draftFreq;
            return;
        }

        const finalUrl = input.trim();

        if (!finalUrl) {
            titleTextarea.value = draftTitle;
            frequencyInput.value = draftFreq;
            return;
        }

        currentLink.link = finalUrl;

        saveLinks();
        updateLinksList();

        displayLink(currentLink);

        titleTextarea.value = draftTitle;
        frequencyInput.value = draftFreq;
    }

    function openLink(link) {
        if (!link) return;

        const title = (link.title || '').trim();

        if (title.startsWith('*')) {
            alert(
                'This link is marked manual-only.\n\n' +
                'Open it manually so it stays outside the app.'
            );
            return;
        }

        const url = link.link?.trim() || '';

        if (url.toLowerCase().startsWith('file://')) {
            alert(
                'This is a local file link.\n\n' +
                'Please open manually:\n\n' +
                url
            );
            return;
        }

        setTimeout(() => {
            window.open(url, '_blank');
        }, 50);
    }

    function startLinks() {
        if (!currentLink) return;
        openLink(currentLink);
        hideStartButton();
    }

    function hideStartButton() {
        if (btnStart) btnStart.style.display = 'none';
    }

    function snoozeCurrentLink() {
        saveCurrent();
        hideStartButton();

        const current = currentLink;
        if (!current) return;

        const d = new Date();
        const freq = (current.frequency || '')
            .toString()
            .trim()
            .toUpperCase();

        if (!freq) {
            d.setDate(d.getDate() + 1);
        } else if (!isNaN(freq)) {
            d.setDate(d.getDate() + parseInt(freq));
        } else if (freq === 'M') {
            d.setMonth(d.getMonth() + 1);
        } else if (freq === 'W') {
            d.setDate(d.getDate() + 7);
        } else if (freq === 'Y') {
            d.setFullYear(d.getFullYear() + 1);
        } else {
            d.setDate(d.getDate() + 1);
        }

        current.snoozed_until = getLocalDateKey(d);

        saveLinks();
        updateLinksList();

        const available = getAvailableLinks();

        if (!available.length) {
            displayNextAvailable();
            return;
        }

        displayLink(available[0]);
        openLink(available[0]);
    }

    function deleteCurrentLink() {
        if (!currentLink) return;

        if (!confirm('Delete this link?')) {
            return;
        }

        saveCurrent();
        hideStartButton();

        const current = currentLink;
        const available = getAvailableLinks();

        let next = null;

        if (available.length > 1) {
            const idx = available.indexOf(current);
            next = available[(idx + 1) % available.length];
        }

        const linkIdx = links.indexOf(current);
        if (linkIdx !== -1) {
            links.splice(linkIdx, 1);
        }

        currentLink = null;

        saveLinks(links.length === 0);
        updateLinksList();

        if (next) {
            displayLink(next);
            openLink(next);
        } else {
            displayNextAvailable();
        }
    }

    async function addFromClipboard() {
        let text = '';
        try {
            if (navigator.clipboard && typeof navigator.clipboard.readText === 'function') {
                text = await navigator.clipboard.readText();
            }
        } catch (e) {}

        let url = (text || '').trim();

        if (
            !url ||
            (
                !url.startsWith('http') &&
                !url.startsWith('file')
            )
        ) {
            url = prompt('Enter URL or file path:');
            if (!url) return;
        }

        const title = prompt(`Enter title for:\n${url}`);
        if (title === null) return;

        links.push({
            id: Date.now(),
            link: url,
            title: title || '',
            frequency: '2',
            snoozed_until: null
        });

        saveLinks();
        displayNextAvailable();
        updateLinksList();
    }

    function updateLinksList() {
        const q = (searchInput.value || '').trim().toLowerCase();

        const filtered = links.filter(l =>
            !q ||
            (l.title || '').toLowerCase().includes(q) ||
            (l.link || '').toLowerCase().includes(q)
        );

        const sorted = [...filtered].sort((a, b) => {
            if (!a.snoozed_until && !b.snoozed_until) {
                return 0;
            }
            if (!a.snoozed_until) {
                return -1;
            }
            if (!b.snoozed_until) {
                return 1;
            }
            return a.snoozed_until.localeCompare(b.snoozed_until);
        });

        linksTableBody.innerHTML = '';
        sorted.forEach(l => {
            const tr = document.createElement('tr');

            const tdSnooze = document.createElement('td');
            tdSnooze.appendChild(document.createTextNode(l.snoozed_until || ''));
            if (l.snoozed_until) {
                const btnClear = document.createElement('button');
                btnClear.textContent = 'x';
                btnClear.onclick = () => clearSnooze(l.id);
                tdSnooze.appendChild(btnClear);
            }

            const tdFreq = document.createElement('td');
            tdFreq.textContent = l.frequency || '';

            const tdLink = document.createElement('td');
            const aLink = document.createElement('a');
            aLink.href = l.link;
            aLink.target = '_blank';
            aLink.textContent = l.title || '(no title)';
            tdLink.appendChild(aLink);
            tdLink.appendChild(document.createElement('br'));
            const smallUrl = document.createElement('small');
            smallUrl.textContent = l.link;
            tdLink.appendChild(smallUrl);

            tr.appendChild(tdSnooze);
            tr.appendChild(tdFreq);
            tr.appendChild(tdLink);
            linksTableBody.appendChild(tr);
        });
    }

    function clearSnooze(id) {
        const l = links.find(x => x.id === id);
        if (l) {
            l.snoozed_until = null;
            saveLinks();
            displayNextAvailable();
            updateLinksList();
        }
    }

    btnLoadFromFile.onclick = loadFromFile;
    btnStart.onclick = startLinks;
    btnSnooze.onclick = snoozeCurrentLink;
    btnDelete.onclick = deleteCurrentLink;
    btnUpdateUrl.onclick = updateCurrentURL;
    btnAddFromClipboard.onclick = addFromClipboard;
    btnOpenReflect.onclick = () => window.open('https://testlisting.pages.dev/reflect.htm', '_blank');
    btnOpen10x.onclick = () => window.open('https://testlisting.pages.dev/10x.htm', '_blank');
    searchInput.oninput = updateLinksList;

    setInterval(saveToFile, 60000);

    window.addEventListener('beforeunload', async () => {
        if (needsFileSave) {
            await saveToFile();
        }
    });

    await init();
})();
