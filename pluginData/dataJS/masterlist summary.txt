    const aliasList = container.querySelector('#alias-list');
    function addAliasRow(canonical = '', aliases = '') {
        const row = document.createElement('div');
        row.className = 'alias-row';

    function getAliasMap() {
        const map = new Map();
        aliasList.querySelectorAll('.alias-row').forEach(row => {
            const canonical = row.querySelector('.canonical').value.trim();
            const aliases = row.querySelector('.aliases').value;
            if (!canonical) return;
            map.set(normaliseForMatch(canonical), canonical);
            aliases.split(',').forEach(a => {
                const n = normaliseForMatch(a);
                if (n) map.set(n, canonical);
            });
        });
        return map;
    }

    function normaliseForMatch(str) {
        return String(str).toLowerCase().replace(/[\s\.\-\/\\,()]/g, '');
    }

    function resolveManufacturer(raw, aliasMap) {
        if (!raw || String(raw).trim() === '') return 'Unknown';
        const key = normaliseForMatch(raw);
        if (aliasMap.has(key)) return aliasMap.get(key);
        for (const [alias, canonical] of aliasMap) {
            if (key.includes(alias) || alias.includes(key)) return canonical;
        }
        return String(raw).trim();
    }

    function parseSheetData(rows, aliasMap) {
        if (!rows || rows.length === 0) return {};
        let mfgIdx = -1;
        let headerIdx = -1;
        for (let i = 0; i < rows.length; i++) {
            const row = rows[i];
            const idx = row.findIndex(h => ['manufacturer', 'make', 'brand', 'supplier'].some(k => String(h).toLowerCase().includes(k)));
            if (idx >= 0) { headerIdx = i; mfgIdx = idx; break; }
        }
        if (headerIdx < 0) return {};
        const counts = {};
        rows.slice(headerIdx + 1).forEach(row => {
            const firstVal = String(row[0]).trim().toLowerCase();
            if (firstVal === 'total' || firstVal === '' || firstVal === 'no.' || firstVal === 'no') return;
            const colF = String(row[5]).trim();
            if (!colF) return;
            const mfg = resolveManufacturer(row[mfgIdx], aliasMap);
            counts[mfg] = (counts[mfg] || 0) + 1;
        });
        return counts;
    }

    function parseTopGamesFromSheets(sheetRowsList, aliasMap) {
        const gameCounts = {};
        sheetRowsList.forEach(rows => {
            if (!rows || rows.length === 0) return;
            let headerIdx = -1;
            let mfgIdx = -1;
            let gameIdx = -1;
            let versionIdx = -1;

            for (let i = 0; i < rows.length; i++) {
                const row = rows[i];
                const mIdx = row.findIndex(h => ['manufacturer', 'make', 'brand', 'supplier', 'mfr'].some(k => String(h).toLowerCase().includes(k)));
                if (mIdx >= 0) {
                    headerIdx = i;
                    mfgIdx = mIdx;
                    gameIdx = row.findIndex(h => {
                        const s = String(h).toLowerCase();
                        return s.includes('game name') || s.includes('game_name') || s.includes('theme') || s.includes('game title') || s === 'game' || s === 'title';
                    });
                    versionIdx = row.findIndex(h => {
                        const s = String(h).toLowerCase();
                        return s.includes('game id') || s.includes('game_id') || s.includes('gameid') || s.includes('version') || s === 'ver' || s.startsWith('ver.') || s.startsWith('ver ');
                    });
                    if (gameIdx < 0) {
                        gameIdx = row.findIndex(h => {
                            const s = String(h).toLowerCase();
                            return s.includes('game');
                        });
                    }
                    break;
                }
            }
            if (headerIdx < 0 || gameIdx < 0) return;

            rows.slice(headerIdx + 1).forEach(row => {
                const firstVal = (row[0] != null ? String(row[0]) : '').trim().toLowerCase();
                if (firstVal === 'total' || firstVal === '' || firstVal === 'no.' || firstVal === 'no' || firstVal === 'undefined') return;

                const gameName = String(row[gameIdx] || '').trim();
                if (!gameName || gameName === 'undefined') return;
                const rawMfg = mfgIdx >= 0 ? row[mfgIdx] : '';
                const mfg = resolveManufacturer(rawMfg, aliasMap);
                const version = versionIdx >= 0 ? String(row[versionIdx] || '').trim() : '';

                const key = `${mfg} | ${gameName} | ${version}`;
                if (!gameCounts[key]) {
                    gameCounts[key] = { mfg, gameName, version, count: 0 };
                }
                gameCounts[key].count += 1;
            });
        });

        const sorted = Object.values(gameCounts).sort((a, b) => b.count - a.count);
        if (sorted.length <= 10) return sorted;

        const cutoffCount = sorted[9].count;
        return sorted.filter((item, idx) => idx < 10 || item.count === cutoffCount);
    }

    function buildGameTable(headers, rows, footerRow) {
        const t = document.createElement('table');
        t.className = 'game-table';
        const thead = t.createTHead();
        const hr = thead.insertRow();
        headers.forEach(h => { const th = document.createElement('th'); th.textContent = h; hr.appendChild(th); });
        const tbody = t.createTBody();
        rows.forEach(row => {
            const tr = tbody.insertRow();
            row.forEach((cell, i) => {
                const td = tr.insertCell();
                td.textContent = (typeof cell === 'number' ? cell.toLocaleString() : cell);
            });
        });
        if (footerRow) {
            const tfoot = document.createElement('tfoot');
            const fr = tfoot.insertRow();
            footerRow.forEach((cell, i) => {
                const td = document.createElement('td');
                td.textContent = (typeof cell === 'number' ? cell.toLocaleString() : cell);
                fr.appendChild(td);
            });
            t.appendChild(tfoot);
        }
        return t;
    }

    function buildTable(headers, rows, footerRow) {
        const t = document.createElement('table');
        const thead = t.createTHead();
        const hr = thead.insertRow();
        headers.forEach(h => { const th = document.createElement('th'); th.textContent = h; hr.appendChild(th); });
        const tbody = t.createTBody();
        rows.forEach(row => {
            const tr = tbody.insertRow();
            row.forEach((cell, i) => {
                const td = tr.insertCell();
                td.textContent = i === 0 ? cell : (typeof cell === 'number' ? cell.toLocaleString() : cell);
                if (headers.length === 4) {
                    if (i === 1) td.className = 'mbs-val';
                    if (i === 2) td.className = 'rws-val';
                    if (i === 3) td.className = 'total-val';
                }
            });
        });
        if (footerRow) {
            const tfoot = document.createElement('tfoot');
            const fr = tfoot.insertRow();
            footerRow.forEach((cell, i) => {
                const td = document.createElement('td');
                td.textContent = i === 0 ? cell : (typeof cell === 'number' ? cell.toLocaleString() : cell);
                if (headers.length === 4) {
                    if (i === 1) td.className = 'mbs-val';
                    if (i === 2) td.className = 'rws-val';
                    if (i === 3) td.className = 'total-val';
                }
                fr.appendChild(td);
            });
            t.appendChild(tfoot);
        }
        return t;
    }

    function createTabbedSection(titleText, tabDefs) {
        const sectionWrapper = document.createElement('div');

        const divTitle = document.createElement('div');
        divTitle.className = 'section-title';
        divTitle.textContent = titleText;
        sectionWrapper.appendChild(divTitle);

        const tabsContainer = document.createElement('div');
        tabsContainer.className = 'tabs';
        const contentContainer = document.createElement('div');

        tabDefs.forEach((def, idx) => {
            const btn = document.createElement('button');
            btn.className = 'tab-btn' + (idx === 0 ? ' active' : '');
            btn.textContent = def.label;
            btn.onclick = () => {
                tabsContainer.querySelectorAll('button').forEach(b => b.classList.remove('active'));
                btn.classList.add('active');
                contentContainer.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));
                const target = contentContainer.querySelector('#tab-' + def.id);
                if (target) target.classList.add('active');
            };
            tabsContainer.appendChild(btn);

            const c = document.createElement('div');
            c.id = 'tab-' + def.id;
            c.className = 'tab-content' + (idx === 0 ? ' active' : '');

            const tabTitle = document.createElement('div');
            tabTitle.className = 'tab-content-title';
            tabTitle.textContent = def.label;
            c.appendChild(tabTitle);

            if (def.element) {
                c.appendChild(def.element);
            } else if (def.render) {
                c.appendChild(def.render());
            }

            contentContainer.appendChild(c);
        });

        sectionWrapper.appendChild(tabsContainer);
        sectionWrapper.appendChild(contentContainer);
        return sectionWrapper;
    }

    async function loadData() {
        if (isLoaded) return;
        const status = container.querySelector("#loadStatus");
        const output = container.querySelector("#outputArea");
        if (!status || !output) return;

        status.textContent = "Loading casino data...";
        output.innerHTML = "";

        try {
            const mbsWB = await PluginAPI.pullData("GRAMBS");
            const rwsWB = await PluginAPI.pullData("GRARWS");
            status.textContent = "✅ Data loaded.";
            toggleBtn.style.display = "inline-block";

            const aliasMap = getAliasMap();
            const EGM_TAB = 'Active EGMs';
            const ETG_TAB = 'Active ETGs';

            const mbsEGM = parseSheetData(mbsWB[EGM_TAB], aliasMap);
            const mbsETG = parseSheetData(mbsWB[ETG_TAB], aliasMap);
            const rwsEGM = parseSheetData(rwsWB[EGM_TAB], aliasMap);
            const rwsETG = parseSheetData(rwsWB[ETG_TAB], aliasMap);

            const mbsEGMTotal = Object.values(mbsEGM).reduce((a,b)=>a+b,0);
            const mbsETGTotal = Object.values(mbsETG).reduce((a,b)=>a+b,0);
            const rwsEGMTotal = Object.values(rwsEGM).reduce((a,b)=>a+b,0);
            const rwsETGTotal = Object.values(rwsETG).reduce((a,b)=>a+b,0);

            const egmOverallTotal = mbsEGMTotal + rwsEGMTotal;
            const etgOverallTotal = mbsETGTotal + rwsETGTotal;
            const grandTotalActive = egmOverallTotal + etgOverallTotal;

            // Table 1: Machine Types
            const div1 = document.createElement('div');
            div1.className = 'section-title';
            div1.textContent = 'Machine Types';
            output.appendChild(div1);
            output.appendChild(buildTable(
                ['Type of Machines', 'MBS', 'RWS', 'Total'],
                [
                    ['Electronic Gaming Machines (EGMs)', mbsEGMTotal, rwsEGMTotal, egmOverallTotal],
                    ['Electronic Table Games (ETGs)',     mbsETGTotal, rwsETGTotal, etgOverallTotal],
                ],
                ['Total', mbsEGMTotal + mbsETGTotal, rwsEGMTotal + rwsETGTotal, grandTotalActive]
            ));

            const merge = (m1, m2) => {
                const keys = new Set([...Object.keys(m1), ...Object.keys(m2)]);
                return Array.from(keys).map(k => ({ name: k, mbs: m1[k]||0, rws: m2[k]||0, total: (m1[k]||0)+(m2[k]||0) })).sort((a,b)=>b.total-a.total);
            };

            // Section 2: Manufacturer Summary (Tabbed: Overall, EGMs, ETGs)
            const allMfgMap = {};
            [mbsEGM, mbsETG, rwsEGM, rwsETG].forEach(m => {
                for (const [mfg, count] of Object.entries(m)) {
                    allMfgMap[mfg] = (allMfgMap[mfg] || 0) + count;
                }
            });
            const sortedOverallMfg = Object.entries(allMfgMap).sort((a, b) => b[1] - a[1]);
            const overallMfgRows = sortedOverallMfg.map(([mfg, count]) => {
                const pct = grandTotalActive > 0 ? (count / grandTotalActive * 100).toFixed(1) : '0.0';
                return [mfg, count, `${pct}%`];
            });
            const overallMfgTable = buildTable(
                ['Manufacturer', 'Count', '% of Total Active Machines'],
                overallMfgRows,
                ['Total', grandTotalActive, '100.0%']
            );

            const egmMerged = merge(mbsEGM, rwsEGM);
            const egmRows = egmMerged.map(r => {
                const pct = egmOverallTotal > 0 ? (r.total / egmOverallTotal * 100).toFixed(1) : '0.0';
                return [r.name, r.mbs, r.rws, `${r.total.toLocaleString()} (${pct}%)`];
            });
            const egmMfgTable = buildTable(
                ['Manufacturer', 'MBS', 'RWS', 'Total (% of Total Active EGMs)'],
                egmRows,
                ['Total', mbsEGMTotal, rwsEGMTotal, `${egmOverallTotal.toLocaleString()} (100.0%)`]
            );

            const etgMerged = merge(mbsETG, rwsETG);
            const etgRows = etgMerged.map(r => {
                const pct = etgOverallTotal > 0 ? (r.total / etgOverallTotal * 100).toFixed(1) : '0.0';
                return [r.name, r.mbs, r.rws, `${r.total.toLocaleString()} (${pct}%)`];
            });
            const etgMfgTable = buildTable(
                ['Manufacturer', 'MBS', 'RWS', 'Total (% of Total Active ETGs)'],
                etgRows,
                ['Total', mbsETGTotal, rwsETGTotal, `${etgOverallTotal.toLocaleString()} (100.0%)`]
            );

            const mfgSection = createTabbedSection('Manufacturer Summary', [
                { id: 'mfg-overall', label: 'Overall', element: overallMfgTable },
                { id: 'mfg-egms',    label: 'EGMs',    element: egmMfgTable },
                { id: 'mfg-etgs',    label: 'ETGs',    element: etgMfgTable }
            ]);
            output.appendChild(mfgSection);

            // Section 3: Top 10 Games (7 Tabs)
            const topGamesTabDefs = [
                {
                    id: 'top-overall',
                    label: 'Overall',
                    sheets: [mbsWB[EGM_TAB], mbsWB[ETG_TAB], rwsWB[EGM_TAB], rwsWB[ETG_TAB]],
                    totalCategoryCount: grandTotalActive,
                    colHeader: 'Count (% of Total Active Machines)'
                },
                {
                    id: 'top-overall-egm',
                    label: 'Overall - EGMs',
                    sheets: [mbsWB[EGM_TAB], rwsWB[EGM_TAB]],
                    totalCategoryCount: egmOverallTotal,
                    colHeader: 'Count (% of Total Active EGMs)'
                },
                {
                    id: 'top-overall-etg',
                    label: 'Overall - ETGs',
                    sheets: [mbsWB[ETG_TAB], rwsWB[ETG_TAB]],
                    totalCategoryCount: etgOverallTotal,
                    colHeader: 'Count (% of Total Active ETGs)'
                },
                {
                    id: 'top-mbs-egm',
                    label: 'MBS - EGMs',
                    sheets: [mbsWB[EGM_TAB]],
                    totalCategoryCount: mbsEGMTotal,
                    colHeader: 'Count (% of Total MBS EGMs)'
                },
                {
                    id: 'top-rws-egm',
                    label: 'RWS - EGMs',
                    sheets: [rwsWB[EGM_TAB]],
                    totalCategoryCount: rwsEGMTotal,
                    colHeader: 'Count (% of Total RWS EGMs)'
                },
                {
                    id: 'top-mbs-etg',
                    label: 'MBS - ETGs',
                    sheets: [mbsWB[ETG_TAB]],
                    totalCategoryCount: mbsETGTotal,
                    colHeader: 'Count (% of Total MBS ETGs)'
                },
                {
                    id: 'top-rws-etg',
                    label: 'RWS - ETGs',
                    sheets: [rwsWB[ETG_TAB]],
                    totalCategoryCount: rwsETGTotal,
                    colHeader: 'Count (% of Total RWS ETGs)'
                }
            ];

            const topGamesTabs = topGamesTabDefs.map(def => {
                const topGames = parseTopGamesFromSheets(def.sheets, aliasMap);
                let tableElement;
                if (topGames.length === 0) {
                    tableElement = document.createElement('div');
                    tableElement.style.padding = '10px';
                    tableElement.textContent = 'No game data available.';
                } else {
                    const rows = topGames.map(g => {
                        const pct = def.totalCategoryCount > 0 ? (g.count / def.totalCategoryCount * 100).toFixed(1) : '0.0';
                        return [g.mfg, g.gameName, g.version || '-', `${g.count.toLocaleString()} (${pct}%)`];
                    });
                    const totalTopCount = topGames.reduce((acc, curr) => acc + curr.count, 0);
                    const totalTopPct = def.totalCategoryCount > 0 ? (totalTopCount / def.totalCategoryCount * 100).toFixed(1) : '0.0';
                    tableElement = buildGameTable(
                        ['Manufacturer', 'Game Name', 'Game ID', def.colHeader],
                        rows,
                        ['Total (Top Games)', '', '', `${totalTopCount.toLocaleString()} (${totalTopPct}%)`]
                    );
                }
                return {
                    id: def.id,
                    label: def.label,
                    element: tableElement
                };
            });

            const topGamesSection = createTabbedSection('Top 10 Active Games', topGamesTabs);
            output.appendChild(topGamesSection);

            isLoaded = true;
        } catch (err) {
            status.textContent = "❌ Error: " + err.message;
        }
    }

})();
